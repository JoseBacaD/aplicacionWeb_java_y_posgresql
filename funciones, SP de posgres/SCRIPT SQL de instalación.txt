CREATE OR REPLACE FUNCTION
	finditem(description varchar, code varchar, department int, currency int, idLocation int, initCode bigint, endCode bigint, idProvider int, idBrand int)
	returns setof item

AS $$
			   

		SELECT distinct on (id_item) id_item,
			   boolean_wildcard,
			   creation_date,
			   desc_item,
			   icon_item_name,
			   id_divition, 
			   id_subdivition, 
			   int_wildcard,
			   item_location,
			   item_stock,
			   last_mod_date,
			   min_quantity, 		   
			   modified_by, 
			   price_sale, 
			   sale_by_fraction,
			   short_desc_item, 
			   standard_code_item, 
			   varchar_wildcard,
			   wholesale_price, 
		    	   id_currency,
			   id_department,
			   id_uom,					
			   min_stock_allowed,
			   id_brand,
			   id_provider,
			   id_location,
			   count(id_item) OVER() as count_rows,
			   price_provider,
  			   id_provider_2,
			   id_provider_3,
			   code_provider,
			   code_provider2,
			   code_provider3,
		           price_provider2,
			   price_provider3
			   
		FROM 
			item 
		WHERE
		
			desc_item ilike '%' || description || '%' OR
			standard_code_item = code OR
			id_department = department OR
			id_currency = currency OR
			id_provider = idProvider OR
			id_brand = idBrand OR
			id_location = idLocation 
			OR id_item BETWEEN initCode and endCode
			

$$ LANGUAGE SQL;

CREATE OR REPLACE FUNCTION
	discountItemStock(headerFolio bigint)
	returns table (id_item int)
AS $$

	update 
		item 
	set 
		item_stock = item_stock - subquery.quantity_item
 	from (select 
				quantity_item, standard_code_item
			from
				sale_detail sd
			where
				folio = headerFolio) AS subquery
				
				
	where item.standard_code_item = subquery.standard_code_item;
		  
select 0 as id_item


$$ LANGUAGE sql;

CREATE OR REPLACE FUNCTION
	findSaleHeader(idDivition int, idSubDiv int,x date, y date, isWithdraw boolean)
RETURNS setof sale_header AS $$

		SELECT 
			  id_sale_header,
			  card_payment,
			  cash_payment,
			  cash_refund,
			  cash_withdraw,
			  comment,
			  consecutive,
			  creation_date,
			  folio,
			  id_divition,
			  id_subdivition,
			  is_withdraw,
			  is_refund,
			  modified_by,
			  other_payment,
			  sale_rate,
			  sale_date,
			  tax_total,
			  total_sale,
			  has_isr_invoice,
			  debit_payment,
			  count(id_sale_header) OVER() as count_rows

		FROM 
			sale_header
		WHERE
			id_divition = idDivition and
			id_subdivition = idSubDiv and
			creation_date between x and y and
			is_withdraw = isWithdraw
		ORDER BY
			folio desc;

$$ LANGUAGE SQL;

CREATE OR REPLACE FUNCTION findSaleDetail(idDivition int, idSubDiv int,x date, y date) 
RETURNS
 setof sale_detail
 AS $$

       SELECT  sd.id_detail
			  ,sd.creation_date
			  ,sd.item_tax
			  ,sd.price_detail
			  ,sd.quantity_item
			  ,sd.sale_date
			  ,sd.folio
			  ,sd.standard_code_item
			  ,count(id_detail) OVER() as count_rows
		FROM 
			sale_detail sd
		WHERE sd.folio in (
			select folio from  sale_header where id_divition = idDivition and
			id_subdivition = idSubDiv and
			creation_date between x and y
			)
			
		ORDER BY
			sd.folio desc;




$$ LANGUAGE sql;

CREATE OR REPLACE FUNCTION
findFrecItem()
	returns TABLE ( id_frecuent_sale_item int,
			   creation_date date,
			   is_frecuent_extra bool,
			   is_frecuent_other bool,
			   is_frecuent_promo bool,
			   standard_code_item varchar
			   )

AS $$
			   

		SELECT subquery.id_frecuent_sale_item,
			   subquery.creation_date,
			   subquery.is_frecuent_extra,
			   subquery.is_frecuent_other,
			   subquery.is_frecuent_promo,
			   subquery.standard_code_item
			  
		FROM 
		
			(select 
			   fsi.id_frecuent_sale_item,
		       fsi.creation_date,
			   fsi.is_frecuent_extra,
			   fsi.is_frecuent_other,
			   fsi.is_frecuent_promo,
			   fsi.standard_code_item,
			   i.desc_item
			 from 
			 	item i, frecuent_sale_item fsi
			 where fsi.standard_code_item = i.standard_code_item) as subquery
			
			order by subquery.desc_item asc
			
			
			
			
			

$$ LANGUAGE SQL;

CREATE OR REPLACE FUNCTION
	findTotalSale(idDivition int, idSubDiv int,x date, y date)
RETURNS TABLE (tot_cash numeric,
			   tot_card numeric,
			   tot_other numeric,
			   grand_total numeric,
		           tot_withdraw numeric,
			   tot_refund numeric,
			   tot_debit numeric
			   ) AS $$

		SELECT 
			sum(cash_payment) as tot_cash,
			sum(card_payment) as tot_card,
			sum(other_payment) as tot_other,
			sum(cash_payment + card_payment + debit_payment + other_payment + cash_withdraw + cash_refund) as grand_total,
			sum(cash_withdraw) as tot_withdraw,
			sum(cash_refund) as tot_refund,
			sum(debit_payment) as tot_debit
		FROM 
			sale_header
		WHERE
			id_divition = idDivition and
			id_subdivition = idSubDiv and
			creation_date between x and y
		

$$ LANGUAGE SQL;
